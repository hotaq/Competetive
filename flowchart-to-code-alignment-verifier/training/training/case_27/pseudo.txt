เริ่มต้น
    รับค่า n
    กำหนดให้ i = 2
    ลูปขณะที่ i <= n แล้ว ทำซ้ำ
        กำหนดให้ isPrime = true
        j = 2
        ลูปขณะที่ j < i แล้ว ทำซ้ำ
            ถ้า i % j == 0 แล้ว
                isPrime = false
                break ออกจากลูป
            j = j + 1
        จบการทำซ้ำ
        ถ้า isPrime = true แล้ว
            แสดงผล i
        i = i + 1
    จบการทำซ้ำ
